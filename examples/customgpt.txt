  /github/pulls:
    get:
      operationId: getPullRequests
      tags:
        - Git Hub Resource
      summary: Retrieve pull requests for a specific repository.
      parameters:
        - name: owner
          in: query
          required: true
          description: The user who owns the repository.
          schema:
            type: string
          example: openjdk
        - name: repo
          in: query
          required: true
          description: The name of the repository.
          schema:
            type: string
          example: jmc
        - name: state
          in: query
          description: "Filter pulls by their state. Can be either `open`, `closed`, or `all`."
          schema:
            type: string
            enum:
              - open
              - closed
              - all
          example: open
        - name: sort
          in: query
          description: "Sorts the results of the pulls. Can be `created`, `updated`, `popularity`, or `long-running`."
          schema:
            type: string
            enum:
              - created
              - updated
              - popularity
              - long-running
          example: created
        - name: direction
          in: query
          description: "The direction of the sort. Can be either `asc` or `desc`."
          schema:
            type: string
            enum:
              - asc
              - desc
          example: desc
        - name: maxresults
          in: query
          description: "The maximum number of results. Defaults to -1 which means as many as we can get. Tips: to get the oldest PR, set maxresults to 1, set sort to created and set the sort order to asc. To get the newest, set the sort order to desc."
          schema:
            type: integer
            minimum: -1
            example: 10
      responses:
        "200":
          description: A list of pull requests for the specified repository.
          content:
            application/json:
              schema:
                type: array
                items:
                  type: object
                  properties:
                    state:
                      type: string
                      description: The state of the pull request (e.g., open, closed).
                    number:
                      type: integer
                      description: The pull request number.
                    title:
                      type: string
                      description: The title of the pull request.
                    created_at:
                      type: string
                      format: date-time
                      description: The date and time when the pull request was created.
                    updated_at:
                      type: string
                      format: date-time
                      description: The date and time when the pull request was last updated.
                    body:
                      type: string
                      description: The body or description of the pull request.
                    user:
                      type: string
                      description: The username of the pull request's author.
